API

Everything renders from a JSON-serializable node tree. Three node kinds: Container, Text, Image. Try it in the playground.

Rust

use hikari::{Node, Style, render_png, render_svg, render_webp};

let tree = Node::container(
    Style::centered()
        .with_size(1200.0, 630.0)
        .with_linear_gradient(180.0, &[(0.0, "#dbeafe"), (1.0, "#fee2e2")]),
    vec![Node::text("Hello", Style::text(84.0, "#0f172a"))],
);
std::fs::write("og.png", render_png(&tree, 1200, 630)?)?;

    title: Some("Invoice".into()),
    paginate: true,   // flow one node across pages
    attachments: vec![Attachment { name: "factur-x.xml".into(), .. }],
    ..PdfOptions::new()
};
let pdf = render_pdf_with(&[doc], PageSize::A4, &options, &License::dev(now_unix()))?;
